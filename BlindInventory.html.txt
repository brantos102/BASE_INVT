    function parsePos(posStr) {
       let parts = posStr.split('-');
       if (parts.length >= 4) {
           let nivel = parseInt(parts.pop()) || 0;
           let col = parseInt(parts.pop()) || 0;
           let literal = parts.pop() || "";
           let nave = parts.join('-') || "";
           return { nave, literal, col, nivel };
       }
       return { nave: posStr, literal: "", col: 0, nivel: 0 };
    }

    function sortHomeTable(key) {
        if (HOME_SORT_CONFIG.key === key) HOME_SORT_CONFIG.asc = !HOME_SORT_CONFIG.asc;
        else { HOME_SORT_CONFIG.key = key; HOME_SORT_CONFIG.asc = true; }
        filterPositions();
    }

    function filterPositionsData() {
        let n = document.getElementById('fltNave')?.value.toUpperCase() || "";
        let r = document.getElementById('fltRack')?.value.toUpperCase() || "";
        let c = document.getElementById('fltCol')?.value.toUpperCase() || "";
        let nv = document.getElementById('fltNiv')?.value.toUpperCase() || "";
        
        return ALL_POS.filter(p => {
            let parts = parsePos(p.nombre);
            return parts.nave.includes(n) && parts.literal.includes(r) && String(parts.col).includes(c) && String(parts.nivel).includes(nv);
        });
    }

    function renderPosList(l) {
       const mapContainer = document.getElementById('rackMapContainer');
       const tableContainer = document.getElementById('posTableContainer');
       
       if(l.length === 0) { 
           mapContainer.innerHTML = "";
           tableContainer.innerHTML = "<div style='color:#999; text-align:center; padding:20px;'>Sin ubicaciones</div>"; 
           return; 
       }

       // 1. MAPA DE RACKS HORIZONTAL (DIBUJA SOLO COLUMNAS FILTRADAS)
       const racks = {};
       l.forEach(p => {
           let parts = parsePos(p.nombre); let rackId = `${parts.nave}-${parts.literal}`;
           if (!rackId || rackId === "-") rackId = "OTROS";
           
           if (!racks[rackId]) racks[rackId] = { maxNiv: 0, cols: new Set(), items: {} };
           if (parts.nivel > racks[rackId].maxNiv) racks[rackId].maxNiv = parts.nivel;
           racks[rackId].cols.add(parts.col); // Solo guarda las columnas que existen en la data filtrada
           
           racks[rackId].items[`${parts.col}-${parts.nivel}`] = p;
       });

       let mapHtml = `<div class="rack-map">`;
       for (let rId in racks) {
           let rack = racks[rId];
           let sortedCols = Array.from(rack.cols).sort((a,b) => a - b);
           
           mapHtml += `<div class="rack-block">
               <div class="rack-title">RACK ${rId}</div>
               <table class="rack-grid">`;
           
           for (let n = rack.maxNiv; n >= 1; n--) {
               mapHtml += `<tr><td class="rack-axis">N${n}</td>`;
               // ITERA SOLO SOBRE LAS COLUMNAS FILTRADAS (No de 1 a maxCol)
               sortedCols.forEach(c => {
                   let posData = rack.items[`${c}-${n}`];
                   if (posData) {
                       let isSel = (posData.nombre === CURRENT_POS) ? "box-shadow: inset 0 0 0 2px black;" : "";
                       mapHtml += `<td class="rack-cell" style="background:${posData.color}; ${isSel}"

                                       onclick="CURRENT_POS='${posData.nombre}'; filterPositions();" 
                                       ondblclick="CURRENT_POS='${posData.nombre}'; nav('count'); loadCountTable();">
                                   </td>`;
                   } else { mapHtml += `<td class="rack-empty"></td>`; }
               });
               mapHtml += `</tr>`;
           }
           mapHtml += `<tr><td></td>`;
           sortedCols.forEach(c => { mapHtml += `<td class="rack-axis">${c}</td>`; });
           mapHtml += `</tr></table></div>`;
       }
       mapHtml += `</div>`;
       mapContainer.innerHTML = mapHtml;

       // 2. TABLA DINÁMICA
       l.sort((a, b) => {
           let pa = parsePos(a.nombre), pb = parsePos(b.nombre);
           let valA, valB;
           switch(HOME_SORT_CONFIG.key) {
               case 'prioridad': valA = a.prioridad; valB = b.prioridad; break;
               case 'avance': valA = a.avance; valB = b.avance; break;
               case 'nombre': valA = a.nombre; valB = b.nombre; break;
               case 'nave': valA = pa.nave; valB = pb.nave; break;
               case 'literal': valA = pa.literal; valB = pb.literal; break;
               case 'col': valA = pa.col; valB = pb.col; break;
               case 'nivel': valA = pa.nivel; valB = pb.nivel; break;
               default: valA = a.prioridad; valB = b.prioridad;
           }
           if (typeof valA === 'string') return HOME_SORT_CONFIG.asc ? valA.localeCompare(valB) : valB.localeCompare(valA);
           else return HOME_SORT_CONFIG.asc ? (valA - valB) : (valB - valA);
       });

       const getSortIcon = (k) => HOME_SORT_CONFIG.key === k ? (HOME_SORT_CONFIG.asc ? '<span class="material-icons sort-indicator" style="font-size:11px;">arrow_drop_up</span>' : '<span class="material-icons sort-indicator" style="font-size:11px;">arrow_drop_down</span>') : '';

               <th onclick="sortHomeTable('nave')">NAVE ${getSortIcon('nave')}</th>
               <th onclick="sortHomeTable('literal')">LIT. ${getSortIcon('literal')}</th>
               <th onclick="sortHomeTable('col')">COL. ${getSortIcon('col')}</th>
               <th onclick="sortHomeTable('nivel')">NIV. ${getSortIcon('nivel')}</th>
           </tr></thead><tbody>`;

       l.forEach(p => {
           let parts = parsePos(p.nombre);
           let isSelected = (p.nombre === CURRENT_POS) ? "background:#e3f2fd;" : "";
           tableHtml += `<tr style="${isSelected}" onclick="CURRENT_POS='${p.nombre}'; filterPositions();" ondblclick="CURRENT_POS='${p.nombre}'; nav('count'); loadCountTable();">
               <td><span style="background:${p.color}; color:white; padding:3px 6px; border-radius:3px; font-size:10px; font-weight:bold; white-space:nowrap;">${p.categoria}</span></td>
               <td><span style="font-size:10px; font-weight:bold;">${p.avance}%</span></td>
               <td style="font-weight:bold; color:var(--primary);">${p.nombre}</td>
               <td>${parts.nave}</td><td>${parts.literal}</td><td>${parts.col}</td><td>${parts.nivel}</td>

       });

    function procesarAcumulacionLocal(item, addQty, qInputStr) {
       if(item.nextStep > 3) return Swal.fire("Full","Fila con 3 conteos cerrados","warning");
       if(item.cat === 'REQUIERE_SERIE' || item.serie) return Swal.fire({ title: 'Requiere Serie', text: '¿Ir a Pestaña Series?', icon: 'info', showCancelButton: true, confirmButtonText: 'Sí' }).then(r => { if (r.isConfirmed) { document.getElementById('targetPos').value = CURRENT_POS; nav('series'); }});

       const inputEl = document.getElementById('qty-' + item.row);
       if (!inputEl) return;
       
       let teorico = parseFloat(item.depot);
       let isManualEdit = qInputStr !== "1" && qInputStr !== ""; 
       
       let currentVal = parseFloat(inputEl.value) || 0; 
       let newVal = isManualEdit ? addQty : currentVal + addQty; 
       
       inputEl.value = newVal; 
       marcarFilaNoGuardada(item.row); 
       inputEl.style.backgroundColor = "#fff8e1";

       if (teorico && isManualEdit && newVal >= teorico) {
           let msg = (newVal === teorico) ? `Se alcanzó la cantidad teórica (${teorico}).` : `Se superó la cantidad teórica (${teorico}). Total escaneado: ${newVal}.`;
           let iconType = (newVal === teorico) ? 'success' : 'warning';
           
           Swal.fire({
               title: 'Confirmar Guardado', text: `${msg} ¿Guardar registro automáticamente?`, icon: iconType, showCancelButton: true, confirmButtonText: 'Sí, guardar', cancelButtonText: 'Seguir contando'
           }).then(r => {
               if(r.isConfirmed) saveRowSpecific(item.row, newVal, document.getElementById('obs-'+item.row)?.value || "");
               else showToast(`+${addQty} (Acumulado: ${newVal})`);
           });
       } else {
           if (teorico && newVal > teorico) showToast(`¡SOBRANTE! Acumulado: ${newVal}`);
           else if (teorico && newVal === teorico) showToast(`¡COMPLETO! Acumulado: ${newVal}`);
           else showToast(`+${addQty} (Total: ${newVal})`);

    function handleSeriesScan(e) {
      if(e.key === 'Enter') {
        let val = e.target.value.trim().toUpperCase(); if(!val) return; e.target.value = ""; 
        
        // ENVÍA EL VALOR TAL COMO ES. EL BACKEND HARÁ EL FUZZY MATCH.
        if(SERIES_DATA.some(i => i.serie === val)) return Swal.fire("Duplicado", "Serie ya escaneada.", "warning");

        const tempId = Date.now(); SERIES_DATA.push({ id: tempId, serie: val, type: 'new', loading: true }); renderSeriesList(); 
        google.script.run.withSuccessHandler(res => {
           const idx = SERIES_DATA.findIndex(i => i.id === tempId);
           if(idx > -1) {
              const item = SERIES_DATA[idx]; item.loading = false;
              
              if (res.serieReal && res.serieReal !== val) item.serie = res.serieReal; 
              
              if(res.duplicado) { item.duplicate = true; item.desc = "DUP: "+(res.datos.pos||"?"); if(res.datos.pos) document.getElementById('displayOrigin').value = res.datos.pos; } 
              else if(res.encontrado) { item.existe = true; item.sku = res.datos.sku; item.desc = res.datos.desc; item.posOriginal = res.datos.pos; document.getElementById('displayOrigin').value = res.datos.pos; let tPos = document.getElementById('targetPos').value.trim().toUpperCase(); if (tPos && res.datos.pos !== tPos) { item.wrongPos = true; Swal.fire({ title: 'Ubicación Diferente', html: `En BD: <b>${res.datos.pos}</b>.<br>Mover a: <b>${tPos}</b>.`, icon: 'warning', timer: 3000 }); } } else { item.existe = false; item.sku = "SIN DATOS"; item.desc = "No registrado"; document.getElementById('displayOrigin').value = "---"; }
              renderSeriesList(); 
           }
        }).validarSerieCompleta(val);
      }
    }

    function renderSeriesList() {
      const div = document.getElementById('seriesList'); div.innerHTML = ""; document.getElementById('cntBadge').innerText = SERIES_DATA.filter(i => i.type === 'new' && !i.duplicate).length;
      SERIES_DATA.slice().reverse().forEach((item, idx) => {
          let b = "4px solid var(--success)", bg = "badge-ok", txt = "NUEVO";
          if(item.loading) { b = "4px solid var(--warning)"; bg = "badge-neutral"; txt = "⏳"; } else if(item.type === 'hist') { b = "4px solid #b0bec5"; bg = "badge-neutral"; txt = "HIST"; } else if(item.duplicate) { b = "4px solid var(--danger)"; bg = "badge-err"; txt = "DUP"; } else if(item.wrongPos) { b = "4px solid var(--orange)"; bg = "badge-err"; txt = "UBIC. DIFERENTE"; } else if(!item.existe) { b = "4px solid var(--warning)"; txt = "?"; }
          div.innerHTML += `<div style="border-left:${b}; padding:8px; border-bottom:1px solid #eee; display:flex; justify-content:space-between; align-items:center; background:${item.type==='hist'?'#fafafa':'white'};"><div><div style="margin-bottom:2px;"><span class="badge ${bg}">${txt}</span> <span class="data-mono" style="font-weight:bold; font-size:14px;">${item.serie}</span></div><div style="font-size:11px; color:#546e7a;">${item.sku||''} ${item.posOriginal?`<span style="color:var(--accent);">[${item.posOriginal}]</span>`:''} <br> ${item.desc||''}</div></div>${item.type === 'new' ? `<button onclick="SERIES_DATA.splice(${SERIES_DATA.length-1-idx},1);renderSeriesList();" style="background:none; color:var(--danger); padding:0;">&times;</button>` : ''}</div>`;
      });
    }

    let PENDING_PAYLOAD = []; 
    function iniciarFlujoProcesoSeries() {
       if(!USER_EMAIL) return Swal.fire("Error", "Falta usuario.", "error");
       if(SERIES_DATA.some(i => i.loading)) return Swal.fire("Espere", "Validando series...", "info");
       const tPos = document.getElementById('targetPos').value.trim().toUpperCase(); if(!tPos) return Swal.fire("Atención", "Defina Destino Final.", "warning");
       const validItems = SERIES_DATA.filter(i => i.type === 'new' && !i.duplicate); if(validItems.length === 0) return Swal.fire("Sin datos", "No hay series válidas.", "info");
       const wrongPos = validItems.filter(i => i.wrongPos);
       if (wrongPos.length > 0) { Swal.fire({ title: 'Cruce de Ubicación', html: `Hay <b>${wrongPos.length}</b> serie(s) de otra ubicación.<br>¿Forzar movimiento a <b>${tPos}</b>?`, icon: 'warning', showCancelButton: true, confirmButtonText: 'Sí, mover' }).then((r) => { if(r.isConfirmed) verificarDesconocidosYGuardar(validItems, tPos); }); return; }
       verificarDesconocidosYGuardar(validItems, tPos);
    }
    function verificarDesconocidosYGuardar(items, tPos) {
       const desc = items.filter(i => !i.existe);
       if (desc.length > 0) { PENDING_PAYLOAD = items; Swal.fire({ title: 'Series Desconocidas', html: `<b>${desc.length}</b> sin registro.<br>¿Reportar Hallazgo en <b>${tPos}</b>?`, icon: 'warning', showCancelButton: true, confirmButtonText: 'Sí' }).then((r) => { if(r.isConfirmed) abrirModalHallazgoBatch(); }); return; }
       addToBackgroundQueue(items, null, tPos);
    }
    function addToBackgroundQueue(items, altData, targetPos) { PROCESSING_QUEUE.push({ items: items, alt: altData, target: targetPos }); SERIES_DATA = SERIES_DATA.filter(i => i.type === 'hist'); renderSeriesList(); document.getElementById('displayOrigin').value = ""; showToast("Procesando fondo..."); if(!IS_PROCESSING_BG) processBackgroundQueue(); }
    function processBackgroundQueue() {
       if(PROCESSING_QUEUE.length === 0) { IS_PROCESSING_BG = false; document.getElementById('syncStatus').style.opacity = 0; return; }
       IS_PROCESSING_BG = true; document.getElementById('syncStatus').style.opacity = 1; 
       const job = PROCESSING_QUEUE.shift(); 
       const payload = job.items.map(i=>({ serie:i.serie, producto:(!i.existe && job.alt) ? job.alt.sku : (i.sku||"SIN_SKU"), posicion: job.target, esDesconocido: !i.existe }));
       google.script.run.withSuccessHandler(res => { if(res.exito) showToast("Series Guardadas"); processBackgroundQueue(); }).withFailureHandler(err => { processBackgroundQueue(); }).procesarGuardadoInventario(payload, job.alt, USER_EMAIL);
    }
